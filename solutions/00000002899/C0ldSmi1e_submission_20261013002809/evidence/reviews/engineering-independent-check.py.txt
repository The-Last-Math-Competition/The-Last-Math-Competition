def read(path):
    return json.loads(path.read_text())

def sha(path):
    h = hashlib.sha256()
    with path.open('rb') as stream:
        for chunk in iter(lambda: stream.read(1048576), b''):
            h.update(chunk)
    return h.hexdigest()

def save(name, value):
    (R / name).write_text(json.dumps(value, indent=2, sort_keys=True) + '\n')
